BEGIN; CREATE TABLE IF NOT EXISTS public.draft_firefighters ( id integer PRIMARY KEY, last_name text NOT NULL UNIQUE, vacation_base integer NOT NULL CHECK (vacation_base >= 0), vacation_added integer NOT NULL DEFAULT 0 CHECK (vacation_added >= 0), accrual_date date, CHECK ((vacation_added = 0 AND accrual_date IS NULL) OR (vacation_added > 0 AND accrual_date IS NOT NULL)) ); CREATE TABLE IF NOT EXISTS public.draft_state ( id integer PRIMARY KEY DEFAULT 1 CHECK (id = 1), round_number integer NOT NULL DEFAULT 1 CHECK (round_number BETWEEN 1 AND 4), current_firefighter_id integer REFERENCES public.draft_firefighters(id), status text NOT NULL DEFAULT 'setup' CHECK (status IN ('setup','active','paused','complete')), turn_started_at timestamptz, updated_at timestamptz NOT NULL DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.draft_selections ( id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY, firefighter_id integer NOT NULL REFERENCES public.draft_firefighters(id), leave_date date NOT NULL, leave_type text NOT NULL CHECK (leave_type IN ('V','K')), round_number integer NOT NULL CHECK (round_number BETWEEN 1 AND 4), created_at timestamptz NOT NULL DEFAULT now(), UNIQUE (firefighter_id, leave_date) ); CREATE INDEX IF NOT EXISTS draft_selections_by_date ON public.draft_selections(leave_date); CREATE TABLE IF NOT EXISTS public.draft_events ( id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY, firefighter_id integer REFERENCES public.draft_firefighters(id), round_number integer CHECK (round_number BETWEEN 1 AND 4), event_type text NOT NULL, details jsonb NOT NULL DEFAULT '{}'::jsonb, created_at timestamptz NOT NULL DEFAULT now() );
